 1  def judge(my_cost, similar):
 2      fair_cost = float(similar["monthly_cost"].median())
 3      raw_pct = round((my_cost - fair_cost) / fair_cost * 100, 9)  # 판정은 반올림 전 값으로 한다
 4      diff_pct = round(raw_pct, 1)
 5      if raw_pct >= 10:
 6          verdict = "비쌈"
 7      elif raw_pct <= -10:
 8          verdict = "저렴"
 9      else:
10          verdict = "적정"
11      low = float(similar["monthly_cost"].min())
12      high = float(similar["monthly_cost"].max())
13      reasons = [
14          f"비슷한 매물 {len(similar)}건의 환산 월 비용 중앙값은 {fair_cost:g}만원입니다.",
15          f"내 매물의 환산 월 비용은 {my_cost:g}만원으로, 적정 비용보다 {diff_pct:+.1f}% 차이가 납니다.",
16          f"비슷한 매물의 환산 월 비용 범위는 {low:g}~{high:g}만원이며, 판정은 '{verdict}'입니다.",
17      ]
18      return {"fair_cost": fair_cost, "diff_pct": diff_pct, "verdict": verdict, "reasons": reasons}
